\chapter{The Kronecker--Weber Theorem}

The Kronecker--Weber theorem asserts that every finite abelian extension of
$\Q$ is contained in a cyclotomic field. We follow the classical reduction:
first to extensions of prime power degree, then to extensions ramified at a
single prime, and finally a case analysis on whether that prime is $2$ or odd.

\section{Statement}

\begin{theorem}[Kronecker--Weber]
  \label{thm:kronecker-weber}
  \lean{KroneckerWeber.abelian_subset_cyclotomic}
  \uses{lem:reduction-prime-power, lem:reduction-single-prime, thm:case-p-2,
    thm:case-odd-p}
  Let $K/\Q$ be a finite abelian extension, that is, a finite normal extension
  whose Galois group $\Gal(K/\Q)$ is abelian. Then there exists a positive
  integer $n$ such that $K \subseteq \Q(\zeta_n)$, where $\zeta_n$ is a
  primitive $n$th root of unity.
\end{theorem}
\begin{proof}
  \uses{lem:reduction-prime-power, lem:reduction-single-prime, thm:case-p-2,
    thm:case-odd-p}
  By Lemma~\ref{lem:reduction-prime-power} it suffices to treat an abelian
  extension $K/\Q$ of degree $p^m$ for a prime $p$. By
  Lemma~\ref{lem:reduction-single-prime} we may further assume that $p$ is the
  only prime that ramifies in $K$. The two remaining cases, $p = 2$ and $p$
  odd, are settled by Theorem~\ref{thm:case-p-2} and
  Theorem~\ref{thm:case-odd-p} respectively, each of which exhibits $K$ inside
  a cyclotomic field. Concretely, by compounding the ramification indices and
  degrees, $K$ is contained in the $nr$th cyclotomic field, where $r$ is the
  product of all primes ramified in $K$ (with an extra factor of $2$ if $2$
  ramifies).
\end{proof}

\section{Reductions}

\begin{lemma}[Compositum of abelian extensions]
  \label{lem:composite-abelian}
  \lean{KroneckerWeber.compositum_abelian}
  If $K/\Q$ and $L/\Q$ are finite abelian extensions, then their compositum
  $KL/\Q$ is again abelian, and the natural map
  \[
    \Gal(KL/\Q) \hookrightarrow \Gal(K/\Q) \times \Gal(L/\Q)
  \]
  is an injective homomorphism.
\end{lemma}
\begin{proof}
  Restriction of automorphisms to $K$ and to $L$ gives a well-defined group
  homomorphism into the direct product. If an element of $\Gal(KL/\Q)$ acts as
  the identity on both $K$ and $L$, it must fix their compositum $KL$
  pointwise, which proves that the map is injective. Since any subgroup of a
  direct product of abelian groups is automatically abelian, it follows that
  $\Gal(KL/\Q)$ is abelian.
\end{proof}

\begin{lemma}[Reduction to prime power degree]
  \label{lem:reduction-prime-power}
  \lean{KroneckerWeber.reduction_prime_power}
  \uses{lem:composite-abelian}
  Suppose the Kronecker--Weber theorem holds for every abelian extension of
  $\Q$ whose degree is a prime power. Then it holds for every finite abelian
  extension of $\Q$.
\end{lemma}
\begin{proof}
  \uses{lem:composite-abelian}
  Let $K/\Q$ be an arbitrary finite abelian extension. By the structure theorem
  for finite abelian groups, the Galois group $\Gal(K/\Q)$ decomposes into a
  direct product of cyclic groups of prime power order. By the fundamental
  theorem of Galois theory, this decomposition implies that $K$ is the
  compositum of a finite family of abelian subextensions $K_i/\Q$, where each
  has prime power degree $[K_i : \Q] = p_i^{m_i}$. Under the hypothesis that the
  theorem holds for all prime power cases, each subfield is contained in some
  cyclotomic field $K_i \subseteq \Q(\zeta_{n_i})$. By
  Lemma~\ref{lem:composite-abelian} their compositum is contained in the
  compositum of these cyclotomic fields. Since
  $\Q(\zeta_{n_1})\,\Q(\zeta_{n_2}) = \Q(\zeta_{\operatorname{lcm}(n_1,n_2)})$,
  we conclude $K \subseteq \Q(\zeta_n)$ with $n = \operatorname{lcm}(n_i)$.
\end{proof}

\begin{lemma}[Tame inertia at $q \neq p$]
  \label{lem:tame-inertia}
  \lean{KroneckerWeber.tame_inertia_cyclic}
  Let $K/\Q$ be abelian of degree $p^m$ and let $q \neq p$ be a prime ramified
  in $K$. For a prime $Q$ of $K$ over $q$ with inertia group $E(Q \mid q)$ and
  ramification index $e = e(Q \mid q)$, the inertia is tame: the first
  ramification group satisfies $V_1(Q \mid q) = \{1\}$, the group $E(Q \mid q)$
  is cyclic, $e$ is a power of $p$, and $e \mid q - 1$.
\end{lemma}
\begin{proof}
  The inertia group $E(Q \mid q)$ is a subgroup of $\Gal(K/\Q)$, so its order
  $e$ is a power of $p$. Because $q \neq p$ equals the residue characteristic
  nowhere relevant here, the ramification is tame, which forces the first
  ramification group $V_1(Q \mid q)$ to be trivial. A tame inertia group is
  cyclic and embeds into the multiplicative group of the residue field, so
  $e \mid q - 1$.
\end{proof}

\begin{lemma}[Unique totally ramified cyclotomic subfield]
  \label{lem:cyclotomic-subfield}
  \lean{KroneckerWeber.cyclotomic_unique_subfield}
  Let $q$ be prime and $e \mid q - 1$. Then the $q$th cyclotomic field
  $\Q(\zeta_q)$ contains a unique subfield $L$ with $[L : \Q] = e$, and $q$ is
  totally ramified in $L$.
\end{lemma}
\begin{proof}
  The group $\Gal(\Q(\zeta_q)/\Q) \cong (\Z/q\Z)^\times$ is cyclic of order
  $q - 1$, so it has a unique subgroup of index $e$ and hence, by the Galois
  correspondence, a unique subfield $L$ with $[L : \Q] = e$. Since $q$ is
  totally ramified in $\Q(\zeta_q)$, it is totally ramified in $L$.
\end{proof}

\begin{lemma}[Stripping one ramified prime via the inertia field]
  \label{lem:inertia-field}
  \lean{KroneckerWeber.inertia_field_strip_prime}
  \uses{lem:composite-abelian, lem:tame-inertia, lem:cyclotomic-subfield}
  In the situation of Lemma~\ref{lem:tame-inertia}, let $L \subseteq \Q(\zeta_q)$
  be the degree $e$ subfield of Lemma~\ref{lem:cyclotomic-subfield}, let $U$ be
  a prime of $KL$ over $Q$, and let $K' = (KL)_{E(U \mid q)}$ be the inertia
  field of $U$ over $q$. Then $q$ is unramified in $K'$, every prime unramified
  in $K$ stays unramified in $K'$, $K'L = KL$, and $[K' : \Q]$ is a power of
  $p$. Consequently $K'$ has strictly fewer ramified primes than $K$, and if
  $K'$ lies in a cyclotomic field then so does $K$.
\end{lemma}
\begin{proof}
  \uses{lem:composite-abelian, lem:tame-inertia, lem:cyclotomic-subfield}
  By the standard properties of inertia fields, $q$ is unramified in
  $K' = (KL)_{E(U \mid q)}$. Any prime of $\Z$ unramified in $K$ stays
  unramified in $KL$, hence in $K'$. The Galois embedding of
  Lemma~\ref{lem:composite-abelian} maps $E(U \mid q)$ into
  $E(Q \mid q) \times \Gal(L/\Q)$. Since $e(U \mid q)$ is a power of $p$ and
  $V_1(U \mid q) = \{1\}$ by Lemma~\ref{lem:tame-inertia}, the group
  $E(U \mid q)$ is cyclic, whence $e(U \mid q) = e$. This equality forces $U$
  to be unramified over $L$ and totally ramified over $K'$, so $K'L = KL$ and
  $[K' : \Q]$ remains a power of $p$. As $q$ is now unramified in $K'$, the set
  of ramified primes strictly shrinks. Finally, if $K' \subseteq \Q(\zeta_N)$
  then $KL = K'L \subseteq \Q(\zeta_{\operatorname{lcm}(N,q)})$, so
  $K \subseteq KL$ lies in a cyclotomic field.
\end{proof}

\begin{lemma}[Reduction to a single ramified prime]
  \label{lem:reduction-single-prime}
  \lean{KroneckerWeber.reduction_single_prime}
  \uses{lem:inertia-field}
  Let $K/\Q$ be abelian of degree $p^m$. If the Kronecker--Weber theorem holds
  for every abelian extension of $p$-power degree in which $p$ is the only
  ramified prime, then it holds for $K$.
\end{lemma}
\begin{proof}
  \uses{lem:inertia-field}
  We induct on the number of primes $q \neq p$ ramified in $K$. If there are
  none, the hypothesis applies directly. Otherwise pick such a $q$ and apply
  Lemma~\ref{lem:inertia-field} to obtain $K'$ of $p$-power degree with $K'L =
  KL$ and strictly fewer ramified primes, such that $K \subseteq \Q(\zeta_n)$
  whenever $K' \subseteq \Q(\zeta_N)$. By the induction hypothesis $K'$ lies in
  a cyclotomic field, hence so does $K$. Iterating eliminates every ramified
  prime $q \neq p$, reducing to the case where $p$ is the unique ramified prime.
\end{proof}

\section{The case \texorpdfstring{$p = 2$}{p = 2}}

\begin{lemma}[Quadratic fields ramified only at $2$]
  \label{lem:quadratic-ram-two}
  \lean{KroneckerWeber.quadratic_ramified_two}
  A quadratic field $K/\Q$ ramified only at $2$ is one of $\Q(\sqrt 2)$,
  $\Q(i)$, or $\Q(\sqrt{-2})$, and each is contained in the $8$th cyclotomic
  field $\Q(\zeta_8)$ via $\sqrt 2 = \zeta_8 + \zeta_8^{-1}$, $i = \zeta_8^2$,
  and $\sqrt{-2} = \zeta_8^3 - \zeta_8$.
\end{lemma}
\begin{proof}
  Writing $K = \Q(\sqrt d)$ with $d$ squarefree, the discriminant of $K$ is $d$
  if $d \equiv 1 \pmod 4$ and $4d$ otherwise. Ramification only at $2$ forces
  the odd part of $d$ to be a unit, so $d \in \{-1, 2, -2\}$, giving $\Q(i)$,
  $\Q(\sqrt 2)$, $\Q(\sqrt{-2})$. The displayed identities exhibit each
  generator inside $\Q(\zeta_8)$.
\end{proof}

\begin{lemma}[Maximal real cyclotomic subfield at $2$]
  \label{lem:real-subfield-two}
  \lean{KroneckerWeber.real_subfield_cyclic_two}
  Let $m > 1$, set $\omega = e^{2\pi i / 2^{m+2}}$, and let $L = \R \cap
  \Q(\omega)$ be the maximal real subfield of $\Q(\zeta_{2^{m+2}})$. Then $L$
  contains $\Q(\sqrt 2)$ as its unique quadratic subfield and $\Gal(L/\Q) \cong
  (\Z/2^{m+2}\Z)^\times / \{\pm 1\}$ is cyclic of order $2^m$.
\end{lemma}
\begin{proof}
  The maximal real subfield $L$ has degree $2^m$ over $\Q$ and Galois group
  $(\Z/2^{m+2}\Z)^\times / \{\pm 1\}$, which is cyclic of order $2^m$ for
  $m > 1$. Its unique quadratic subfield is the real field $\Q(\sqrt 2)$.
\end{proof}

\begin{theorem}[Kronecker--Weber for $p = 2$]
  \label{thm:case-p-2}
  \lean{KroneckerWeber.case_two}
  \uses{lem:composite-abelian, lem:quadratic-ram-two, lem:real-subfield-two}
  Let $K/\Q$ be abelian of degree $2^m$ in which $2$ is the only ramified
  prime. Then $K$ is contained in a cyclotomic field.
\end{theorem}
\begin{proof}
  \uses{lem:composite-abelian, lem:quadratic-ram-two, lem:real-subfield-two}
  For $m = 1$ the field $K$ is quadratic and ramified only at $2$, so by
  Lemma~\ref{lem:quadratic-ram-two} it sits inside $\Q(\zeta_8)$.

  For $m > 1$, $K$ contains $\Q(\sqrt 2)$ as its unique real quadratic
  subfield. Take $L = \R \cap \Q(\omega)$ with $\omega = e^{2\pi i/2^{m+2}}$ as
  in Lemma~\ref{lem:real-subfield-two}, so $\Gal(L/\Q)$ is cyclic of order
  $2^m$. Let $\sigma$ generate $\Gal(L/\Q)$ and extend it to an automorphism
  $\tau$ of the compositum $KL$ (Lemma~\ref{lem:composite-abelian}); let $F$ be
  the fixed field of $\tau$. Since $\tau$ restricts to $\sigma$ on $L$, any
  element of $F \cap L$ is fixed by $\sigma$, so $F \cap L = \Q$. Hence $F$ is a
  quadratic extension of $\Q$ ramified only at $2$, i.e. $F \in \{\Q, \Q(i),
  \Q(\sqrt{-2})\}$, and $\tau$ has order exactly $2^m$. Matching degrees and
  ramification profiles forces $K \subseteq \Q(\omega)$, placing $K$ inside a
  cyclotomic field.
\end{proof}

\section{The case of odd \texorpdfstring{$p$}{p}}

\begin{lemma}[Different exponent for $m = 1$, odd $p$]
  \label{lem:odd-different-base}
  \lean{KroneckerWeber.odd_different_base}
  Let $p$ be odd and $K/\Q$ of degree $p$ in which $p$ is totally ramified,
  with prime $P$ over $p$. Then the different is
  $\operatorname{diff}(\mathcal{O}_K \mid \Z) = P^{2(p-1)}$.
\end{lemma}
\begin{proof}
  Since $p$ is totally ramified, $e(P \mid p) = p$. Choose a uniformizer
  $\pi \in P \setminus P^2$; it generates $K$ with degree $p$ over $\Q$, and
  its monic minimal polynomial $f(x) = x^p + a_1 x^{p-1} + \dots + a_p$ has all
  $a_i$ divisible by $p$. Writing $P^k \,\|\, f'(\pi)$, a ramification argument
  shows $p - 1 \mid k$. Expanding $f'(\pi) = p\pi^{p-1} + a_1(p-1)\pi^{p-2} +
  \dots + a_{p-1}$, the $P$-adic valuations of the terms are pairwise
  incongruent modulo $p$, hence distinct, so $k$ equals their minimum, which is
  $2(p-1)$. Therefore $\operatorname{diff}(\mathcal{O}_K \mid \Z) = P^{2(p-1)}$.
\end{proof}

\begin{lemma}[Cyclicity of the Galois group, odd $p$]
  \label{lem:odd-galois-cyclic}
  \lean{KroneckerWeber.odd_galois_cyclic}
  \uses{lem:odd-different-base}
  Let $p$ be odd and $K/\Q$ abelian of degree $p^2$ in which $p$ is the only
  ramified prime. Then $G = \Gal(K/\Q)$ is cyclic.
\end{lemma}
\begin{proof}
  \uses{lem:odd-different-base}
  The prime $p$ is totally ramified, so the inertia group $E(P \mid p)$ and the
  first ramification group $V_1$ both have order $p^2$. Let $V_r$ be the first
  lower ramification group whose order drops below $p^2$; then $|V_r| = p$. For
  a subgroup $H \subset G$ of order $p$ with fixed field $K_H$, the tower
  formula for differents gives $\operatorname{diff}(\mathcal{O}_K \mid \Z) =
  \operatorname{diff}(\mathcal{O}_K \mid \mathcal{O}_{K_H})\, P^{2(p-1)p}$, using
  Lemma~\ref{lem:odd-different-base}. By Hilbert's different formula $k =
  \sum_{m \ge 0}(|V_m| - 1)$, the exponent of $P$ in
  $\operatorname{diff}(\mathcal{O}_K \mid \mathcal{O}_{K_H})$ is uniquely
  maximized when $H = V_r$. If $G$ were elementary abelian $\Z/p \times \Z/p$,
  all order-$p$ subgroups would be symmetric and no single $H$ could uniquely
  maximize the exponent, a contradiction. Hence $G$ is cyclic of order $p^2$.
\end{proof}

\begin{lemma}[Unique cyclotomic subfield of $p$-power degree, odd $p$]
  \label{lem:odd-unique-subfield}
  \lean{KroneckerWeber.odd_unique_subfield}
  Let $p$ be odd and $m \ge 1$. The $p^{m+1}$th cyclotomic field
  $\Q(\zeta_{p^{m+1}})$ has a unique subfield $L$ with $[L : \Q] = p^m$, and
  $\Gal(L/\Q)$ is cyclic of order $p^m$.
\end{lemma}
\begin{proof}
  The group $\Gal(\Q(\zeta_{p^{m+1}})/\Q) \cong (\Z/p^{m+1}\Z)^\times$ is cyclic
  of order $p^m(p-1)$, so it has a unique subgroup of index $p^m$ and hence a
  unique subfield $L$ of degree $p^m$, with $\Gal(L/\Q)$ cyclic of order $p^m$.
\end{proof}

\begin{theorem}[Kronecker--Weber for odd $p$]
  \label{thm:case-odd-p}
  \lean{KroneckerWeber.case_odd}
  \uses{lem:composite-abelian, lem:odd-galois-cyclic, lem:odd-unique-subfield}
  Let $p$ be an odd prime and let $K/\Q$ be abelian of degree $p^m$ in which
  $p$ is the only ramified prime. Then $K$ is contained in a cyclotomic field.
\end{theorem}
\begin{proof}
  \uses{lem:composite-abelian, lem:odd-galois-cyclic, lem:odd-unique-subfield}
  The cyclicity established in Lemma~\ref{lem:odd-galois-cyclic} (and its
  general-$m$ analogue) pins down the structure of $K$. Let $L$ be the unique
  degree $p^m$ subfield of $\Q(\zeta_{p^{m+1}})$ from
  Lemma~\ref{lem:odd-unique-subfield}, with $\Gal(L/\Q)$ cyclic of order $p^m$.
  Let $\sigma$ generate $\Gal(L/\Q)$, extend it to an automorphism $\tau$ of
  $KL$ (Lemma~\ref{lem:composite-abelian}), and let $F$ be the fixed field of
  $\tau$. Then $F \cap L = \Q$, and the uniqueness of extensions ramified only
  at $p$ forces $F = \Q$. Therefore $K = L \subseteq \Q(\zeta_{p^{m+1}})$, so
  $K$ lies in a cyclotomic field.
\end{proof}
